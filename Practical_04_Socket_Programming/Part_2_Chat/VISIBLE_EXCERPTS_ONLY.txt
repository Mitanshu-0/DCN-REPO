import socket

s = socket.socket()
s.bind(("0.0.0.0", 12345))
s.listen(1)

c, addr = s.accept()
print("Connected:", addr)

while True:
    msg = c.recv(1024).decode()
    print("Client:", msg)
    if msg == "bye":
        break

    msg = input("Server: ")
    c.sendall(msg.encode())
    if msg == "bye":
        break

c.close()
s.close()
